<script>



// =======================================================

// SETTINGS

// =======================================================



// Replace this with YOUR Apps Script deployment URL.


const APPS_SCRIPT_URL =

  "https://script.google.com/a/macros/g.ecc.u-tokyo.ac.jp/s/AKfycbxTb_jZj55UNHbv3fcv-CiAJnNRqHFFZCeyFAQViqqqNkYCC9hNCddLe1qmNIqAoXFE/exec";



// Scanner is disabled for this many milliseconds

// after a successful scan.


const COOLDOWN_MS = 5000;



// =======================================================

// VARIABLES

// =======================================================


let currentTable = "";


let scanner = null;


let scanLocked = false;



// =======================================================

// TABLE SETUP

// =======================================================


function setTable() {


  const input =

    document.getElementById("tableInput");


  const name =

    input.value.trim();



  if (!name) {


    alert("Please enter a table name.");


    return;

  }



  currentTable = name;



  document.getElementById("currentTable")

    .textContent = currentTable;



  document.getElementById("setupScreen")

    .style.display = "none";



  // Start camera only the first time.


  if (!scanner) {


    startScanner();


  }


}




function changeTable() {


  document.getElementById("tableInput").value =

    currentTable;


  document.getElementById("setupScreen")

    .style.display = "flex";


}




// =======================================================

// QR SCANNER

// =======================================================


function startScanner() {


  scanner =

    new Html5Qrcode("reader");



  const config = {


    fps: 10,


    qrbox: {

      width: 250,

      height: 250

    }


  };



  scanner.start(


    {

      facingMode: "environment"

    },


    config,


    onScanSuccess,


    onScanFailure


  ).catch(function(error) {


    console.error(error);


    alert(

      "Could not access the camera. " +

      "Please allow camera permission and reload the page."

    );


  });


}




// =======================================================

// SUCCESSFUL QR DETECTION

// =======================================================


function onScanSuccess(decodedText) {



  // Ignore scans while cooldown is active.


  if (scanLocked) {

    return;

  }



  scanLocked = true;



  // Send data to Google Sheet.


  sendToGoogleSheet(

    decodedText,

    currentTable

  );



  // Show thank-you screen.


  const message =

    document.getElementById("message");


  message.style.display = "flex";



  // Wait 5 seconds, then return to scanner.


  setTimeout(function() {


    message.style.display = "none";


    scanLocked = false;


  }, COOLDOWN_MS);


}




// =======================================================

// IGNORE SCANNER FAILURES

// =======================================================


function onScanFailure(error) {


  // A QR scanner generates many "not found" messages

  // while looking at the camera image.

  // We intentionally do nothing here.


}




// =======================================================

// SEND DATA TO GOOGLE APPS SCRIPT

// =======================================================


function sendToGoogleSheet(qrText, tableName) {



  // We submit a normal HTML form to Apps Script.

  //

  // This avoids browser cross-origin restrictions.

  //

  // The response goes into a hidden iframe, so

  // the visitor never leaves the scanner page.



  const form =

    document.createElement("form");



  form.method = "POST";


  form.action =

    APPS_SCRIPT_URL;


  form.target =

    "googleResponse";


  form.style.display =

    "none";




  // QR value


  const qrField =

    document.createElement("input");


  qrField.name =

    "qr";


  qrField.value =

    qrText;



  // Table name


  const tableField =

    document.createElement("input");


  tableField.name =

    "table";


  tableField.value =

    tableName;



  form.appendChild(qrField);


  form.appendChild(tableField);



  document.body.appendChild(form);



  form.submit();



  // Remove temporary form.


  setTimeout(function() {


    form.remove();


  }, 1000);


}



</script>
